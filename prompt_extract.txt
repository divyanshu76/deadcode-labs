function parseColor(input: string): RGB {
    const s = (input || "").trim()
    if (s.charAt(0) === "#") {
        let h = s.slice(1)
        if (h.length === 3 || h.length === 4) {
            h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
        }
        const n = parseInt(h.slice(0, 6), 16)
        if (!isFinite(n)) return [255, 255, 255]
        return [(n >> 16) & 255, (n >> 8) & 255, n & 255]
    }
    const m = s.match(/-?\d*\.?\d+/g)
    if (m && m.length >= 3) {
        return [Number(m[0]), Number(m[1]), Number(m[2])]
    }
    return [255, 255, 255]
}

function rampAt(stops: RGB[], u: number, out: RGB) {
    const n = stops.length
    if (n === 1) {
        out[0] = stops[0][0]
        out[1] = stops[0][1]
        out[2] = stops[0][2]
        return
    }
    const x = u - Math.floor(u)
    const f = x * n
    const i0 = Math.floor(f) % n
    const i1 = (i0 + 1) % n
    let k = f - Math.floor(f)
    k = k * k * (3 - 2 * k)
    const a = stops[i0]
    const b = stops[i1]
    out[0] = a[0] + (b[0] - a[0]) * k
    out[1] = a[1] + (b[1] - a[1]) * k
    out[2] = a[2] + (b[2] - a[2]) * k
}

function clamp(v: number, lo: number, hi: number) {
    return v < lo ? lo : v > hi ? hi : v
}

function rgbString(r: number, g: number, b: number, a: number) {
    return (
        "rgba(" +
        Math.round(clamp(r, 0, 255)) +
        "," +
        Math.round(clamp(g, 0, 255)) +
        "," +
        Math.round(clamp(b, 0, 255)) +
        "," +
        clamp(a, 0, 1).toFixed(3) +
        ")"
    )
}

function knot(
    t: number,
    a: number,
    b: number,
    p: Float64Array,
    off: number
) {
    const s1 = Math.sin(t)
    const c1 = Math.cos(t)
    const s2 = Math.sin(2 * t)
    const c2 = Math.cos(2 * t)
    const s3 = Math.sin(3 * t)
    const c3 = Math.cos(3 * t)
    p[off] = s1 + a * s2
    p[off + 1] = c1 - a * c2
    p[off + 2] = -b * s3
    p[off + 3] = c1 + 2 * a * c2
    p[off + 4] = -s1 + 2 * a * s2
    p[off + 5] = -3 * b * c3
}

function ellipseAxes(
    a: number,
    b: number,
    c: number,
    d: number,
    out: Float64Array
) {
    const E = (a + d) * 0.5
    const F = (a - d) * 0.5
    const G = (b + c) * 0.5
    const H = (b - c) * 0.5
    const Q = Math.hypot(E, H)
    const R = Math.hypot(F, G)
    out[0] = Q + R
    out[1] = Math.abs(Q - R)
    out[2] = (Math.atan2(H, E) + Math.atan2(G, F)) * 0.5
}
